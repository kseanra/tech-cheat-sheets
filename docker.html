Docker CLI

Essential Docker commands and quick examples to get containers running locally.

docker-basics.sh cli 
# show Docker version
docker --version

# test installation (runs a tiny container that prints a message)
docker run hello-world

# pull an image from Docker Hub
docker pull nginx:latest

# list images on the local machine
docker images

# list running containers
docker ps

# list all containers (including stopped)
docker ps -a

# run a container in the background, map port 80
docker run -d -p 8080:80 --name web nginx:latest

# run an interactive shell in a new container and remove when exited
docker run --rm -it alpine sh

# run a command in a running container (get a shell)
docker exec -it web /bin/bash

# view live logs for a container
docker logs -f web

# stop / start a container
docker stop web

docker start web

# remove a stopped container
docker rm web

# remove an image
docker rmi nginx:latest

# build an image from a Dockerfile in the current dir
docker build -t myapp:latest .

# helpful cleanup: remove dangling images
docker image prune -f
